```
#include<stdio.h>
void print_stars() { //아무것도 입력받지 않는다
	for (int i = 0; i < 30; i++) // *을 30번 출력한다
		printf("*");
}
int main(void) {
	print_stars(); //함수 실행
	printf("\nHello World!\n");
	print_stars(); //함수 실행
	printf("\n");
	for (size_t i = 0; i < 100; i++) 
	{
		print_stars(); //함수를 100번 반복해 실행
		printf("\n");
	}
	return 0;

}
```

#include<stdio.h>
int max(int x, int y) { /max라는 함수, (x, y)를 입력받는다. 입출력은 모두 int
	if (x > y) //x가 y보다 큰가?
		return x; //x값 반환
	else
		return y; //y값 반환
}

int main() {
	int a, b;
	scanf("%d%d", &a,&b);
	printf("%d", max(a, b)); // (x, y)형식

}
